db.orders.aggregate([
    {
        $group: {
            _id: "$status",
            count: { $sum: 1 }
        }
    },
    {
        $sort: { count: -1 }
    }
])

db.orders.aggregate([
    {
        $match: { status: "delivered" }
    },
    {
        $group: {
            _id: "$item",
            totalQty: { $sum: "$qty" },
            revenue: { $sum: { $multiply: ["$qty", "$price"] } }
        }
    }
])


db.orders.aggregate([
    {
        $match: { status: "delivered" }
    },
    {
        $project: {
            _id: 0,
            customer: 1,
            item: 1,
            qty: 1
        }
    }
])


db.orders.aggregate([
    {
        $match: { status: "delivered" }
    },
    {
        $group: {
            _id: "$customer",
            orders: { $sum: 1 },
            totalSpend: {
                $sum: { $multiply: ["$qty", "$price"] }
            }
        }
    },
    {
        $sort: { totalSpend: -1 }
    }
])
